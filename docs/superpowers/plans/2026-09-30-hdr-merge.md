# HDR Bracket Merge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Merge DNG exposure brackets into one floating-point raw photo, carry its brighter-than-white range through the raw path, and roll it off in the tone controls, so it edits like any photo with recovered highlights.

**Architecture:** `hdrmerge.js` (pure, UMD) finds bracket sets and merges CFA planes into a `Float32Array` in reference-exposure units (values up to `hdrMax`). `source.js` builds a raw descriptor from a set (`decodeMerged`). The GPU raw path takes float planes (`R32F`, float shader variants) and clamps at `uRawMax` instead of 1; the tone step (`edge.js` applyTone ⇄ `TONE_GLSL`) gains a highlight shoulder that is active only when the headroom exceeds 1, so ordinary files are bit-identical. The strip marks sets and adds merged entries.

**Tech Stack:** Plain browser JS (UMD modules, no build), WebGL2, node:test (`npm test`), headless Chrome via scratchpad `cdp.py` + `nocache.py` (serve the repo on 8766).

**Spec:** `docs/superpowers/specs/2026-09-30-hdr-merge-design.md`

- Merge constants: `KNEE = 0.80`, `CLIP = 0.95` (block max, normalised), noise weight `t_i`, reference = median exposure (even count: the longer middle one).
- Shoulder: knee `K = 0.75`; headroom `H = srgbEncode(rawMax) · 2^exposureEv`; active only when `H > 1`; formula `l' = K + (1−K)·x(1 + x/X²)/(1 + x)`, `x = (l−K)/(1−K)`, `X = (H−K)/(1−K)`.

- Parity tolerances: float vs 16-bit raw ≤ 1e-3; tone GPU vs JS as the existing check (5e-4).

```js
(function (root, factory) {
  'use strict';
  const api = factory();
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  else root.HdrMerge = api;
})(typeof globalThis !== 'undefined' ? globalThis : this, function () {
  'use strict';

  // Merging raw exposure brackets (a still camera, a still scene) into one
  // floating-point CFA plane in the reference frame's units: 1 is the
  // reference frame's white, and the plane reaches hdrMax = t_ref / t_min
  // where only the shortest frame held the light. Per 2x2 Bayer block, each
  // frame counts with weight t_i (a longer exposure collected more light, so
  // it is cleaner), faded out as the block nears clipping in that frame --
  // per block, so a pixel's channels never come from different frames
  // (no magenta highlights). Blocks clipped in every frame take the
  // shortest frame.
  const KNEE = 0.80, CLIP = 0.95;

  const second = (m) => String(m.dateTimeOriginal || '').slice(0, 19);
  const sameShot = (a, b) => a.model === b.model && second(a) === second(b) && a.iso === b.iso &&
    a.width === b.width && a.height === b.height && String(a.cfaPattern) === String(b.cfaPattern);

  function detectSets(items) {
    const sets = [], used = new Set();
    for (const it of items) {
      if (used.has(it.id) || !it.meta || !(it.meta.exposureTime > 0) || !second(it.meta)) continue;
      const group = [it];
      for (const o of items) {
        if (o === it || used.has(o.id) || !o.meta || !(o.meta.exposureTime > 0)) continue;
        if (!sameShot(it.meta, o.meta)) continue;
        if (group.some((g) => Math.abs(g.meta.exposureTime - o.meta.exposureTime) < 1e-9)) continue;
        group.push(o);
      }
      if (group.length < 2) continue;
      for (const g of group) used.add(g.id);
      sets.push(group.sort((a, b) => a.meta.exposureTime - b.meta.exposureTime).map((g) => g.id));
    }
    return sets;
  }

  function canMerge(metas) {
    if (metas.length < 2) return { ok: false, reason: 'Select two or more raw photos of one bracket.' };
    const a = metas[0];
    for (const m of metas) {
      if (!(m.exposureTime > 0)) return { ok: false, reason: 'A photo has no exposure time.' };
      if (m.model !== a.model) return { ok: false, reason: 'The photos are from different cameras.' };
      if (m.width !== a.width || m.height !== a.height || String(m.cfaPattern) !== String(a.cfaPattern)) {
        return { ok: false, reason: 'The photos differ in size.' };
      }
      if (m.iso !== a.iso) return { ok: false, reason: 'The photos differ in ISO.' };
    }
    const times = new Set(metas.map((m) => m.exposureTime.toPrecision(6)));
    if (times.size !== metas.length) return { ok: false, reason: 'Two photos have the same exposure.' };
    return { ok: true, reason: '' };
  }

  const mergeKey = (keys) => 'HDR|' + [...keys].sort().join('||');

  function referenceIndex(times) {
    const order = times.map((t, i) => i).sort((a, b) => times[a] - times[b]);
    return order[Math.floor(times.length / 2)];
  }

  const smoothstep = (a, b, x) => { const t = Math.min(1, Math.max(0, (x - a) / (b - a))); return t * t * (3 - 2 * t); };

  function mergePlanes(frames, w, h) {
    const times = frames.map((f) => f.t);
    const ref = referenceIndex(times);
    const tRef = times[ref];
    let shortest = 0;
    for (let i = 1; i < frames.length; i++) if (times[i] < times[shortest]) shortest = i;
    const hdrMax = tRef / times[shortest];
    const out = new Float32Array(w * h);
    const n = frames.length;
    const scale = frames.map((f) => tRef / (f.t * (f.white - f.black)));
    const idx = [0, 0, 0, 0];
    const wsum = new Float64Array(n);
    for (let by = 0; by < h; by += 2) {
      for (let bx = 0; bx < w; bx += 2) {
        idx[0] = by * w + bx; idx[1] = idx[0] + 1; idx[2] = idx[0] + w; idx[3] = idx[2] + 1;
        const cnt = (bx + 1 < w ? 2 : 1) * (by + 1 < h ? 2 : 1) === 4 ? 4 : null;
        let total = 0;
        for (let i = 0; i < n; i++) {
          const f = frames[i];
          let m = 0;
          for (let k = 0; k < 4; k++) {
            if (cnt === null && ((k & 1 && bx + 1 >= w) || (k & 2 && by + 1 >= h))) continue;
            const v = (f.plane[idx[k]] - f.black) / (f.white - f.black);
            if (v > m) m = v;
          }
          wsum[i] = (1 - smoothstep(KNEE, CLIP, m)) * f.t;
          total += wsum[i];
        }
        for (let k = 0; k < 4; k++) {
          if (cnt === null && ((k & 1 && bx + 1 >= w) || (k & 2 && by + 1 >= h))) continue;
          const p = idx[k];
          let r;
          if (total > 0) {
            r = 0;
            for (let i = 0; i < n; i++) if (wsum[i] > 0) r += wsum[i] * (frames[i].plane[p] - frames[i].black) * scale[i];
            r /= total;
          } else {
            const f = frames[shortest];
            r = (f.plane[p] - f.black) * scale[shortest];
          }
          out[p] = Math.max(0, r);
        }
      }
    }
    return { plane: out, hdrMax, ref };
  }

  return { KNEE, CLIP, detectSets, canMerge, mergeKey, referenceIndex, mergePlanes };
});
```

  // Brighter than white (a merged HDR photo, rawMax > 1): lightness above
  // SHOULDER_KNEE is compressed so the headroom H -- the lightness a fully
  // clipped white reaches after Exposure -- lands exactly on 1. Slope 1 at

  const SHOULDER_KNEE = 0.75;
  function shoulder(l, H) {
    const K = SHOULDER_KNEE;
    if (H <= 1 || l <= K) return l;
    const x = (l - K) / (1 - K), X = (H - K) / (1 - K);
    return K + (1 - K) * x * (1 + x / (X * X)) / (1 + x);
  }
